4. Buttons use signals

A signal is a notification from a node. A button sends pressed when activated. connect() tells Godot which function should respond.

$Menu/Play.pressed.connect(start_run)
$GameOverPanel/Retry.pressed.connect(start_run)
$Home.pressed.connect(show_screen.bind("menu"))

The first two buttons both call start_run(). The Home connection uses bind("menu") to supply the screen name when the button calls show_screen().

These connections happen once inside _ready(). The paths match the names in the Scene panel exactly; capitalization matters.

Shooting now uses _unhandled_input(). Godot gives buttons the first chance to handle a click. That prevents Play, Retry or Home clicks from also firing the comet. A released mouse button or lost window focus cancels a stale aim.

5. When does “Out of energy” appear?

The smallest accepted shot has speed 60. Maximum shot speed is 630. Using your cost formula, the cheapest valid shot costs approximately 11.90 energy.

const MIN_SHOT_COST: float = 10.0 + 20.0 * 60.0 / 630.0

Testing only for exactly zero would leave a player stuck at 10 energy. The loss check asks whether energy is below the minimum valid cost and the three-second heat timer has ended.

if lit_count == planets.size():
    show_result(true)
elif hot_time <= 0.0 and energy + 0.000001 < MIN_SHOT_COST:
    show_result(false)

The tiny extra number protects against decimal rounding at the exact minimum. The check happens after real relights, so the last hot shot can still earn 15 energy. Winning is checked first, so lighting the final planet wins even when energy is low.

The same panel displays the win result with different text and a Play Again button.

func show_result(won: bool) -> void:
	run_active = false
	$GameOverPanel/Title.text = "All planets lit!" if won else "Out of energy"
	$GameOverPanel/Message.text = "You brought light back to all four worlds." if won else "Your last shot has cooled. Try a fresh run."
	$GameOverPanel/Retry.text = "Play Again" if won else "Retry"
	show_screen("won" if won else "lost")

func _physics_process(delta: float) -> void:
	if screen != "play":
		return
	if is_aiming and not Input.is_mouse_button_pressed(MOUSE_BUTTON_LEFT):
		is_aiming = false
	hot_time = maxf(hot_time - delta, 0.0)
	move_comet(delta)

	if has_launched and hot_time > 0.0:
		for planet in planets:
			if planet["lit"]:
				continue

			var planet_node: Sprite2D = planet["node"]
			var radius: float = planet["radius"]
			var distance: float = comet_position.distance_to(planet_node.position)

			if distance <= radius + 20.0 + 0.1:
				planet["lit"] = true
				planet_node.modulate = Color.WHITE
				energy = minf(energy + 15.0, 100.0)
	var lit_count: int = planets.filter(func(planet: Dictionary) -> bool: return planet["lit"]).size()
	$PlanetsLabel.text = "Planets: %d / %d" % [lit_count, planets.size()]
	if lit_count == planets.size():
		show_result(true)
	elif hot_time <= 0.0 and energy + 0.000001 < MIN_SHOT_COST:
		show_result(false)
	$EnergyLabel.text = "Energy: %.1f / 100" % energy
	$EnergyBar.value = energy
	if is_aiming:
		var shot_velocity: Vector2 = ((drag_start - get_local_mouse_position()) * 4.0).limit_length(630.0)

		if shot_velocity.length() >= 60.0:
			var power: float = shot_velocity.length() / 630.0
			var shot_cost: float = 10.0 + 20.0 * power
			$EnergyLabel.text += "   |   Shot cost: %.1f" % shot_cost

			if energy < shot_cost:
				$EnergyLabel.text += "   Not enough energy"
		else:
			$EnergyLabel.text += "   |   Drag farther to shoot"
	queue_redraw()

func preview_path(launch_velocity: Vector2) -> PackedVector2Array:
	var saved_position: Vector2 = comet_position
	var saved_velocity: Vector2 = comet_velocity
	var saved_launched: bool = has_launched

	var points := PackedVector2Array()
	var step: float = 1.0 / Engine.physics_ticks_per_second

	comet_velocity = launch_velocity
	has_launched = true

	for dot_index in range(20):
		for small_step in range(3):
			move_comet(step)
		points.append(comet_position)

	comet_position = saved_position
	comet_velocity = saved_velocity
	has_launched = saved_launched

	return points

func move_comet(delta: float) -> void:
	if has_launched:
		var nearest_distance: float = INF

		for planet in planets:
			var planet_node: Sprite2D = planet["node"]
			var radius: float = planet["radius"]
			var to_planet: Vector2 = planet_node.position - comet_position
			var distance: float = to_planet.length()

			nearest_distance = minf(nearest_distance, distance)

			var size_factor: float = radius / 60.0
			var strength: float = gravity_strength * size_factor * size_factor
			var safe_distance: float = maxf(distance, 40.0)
			var pull: float = strength / (safe_distance * safe_distance)

			comet_velocity += to_planet.normalized() * pull * delta

		var far_amount: float = clampf((nearest_distance - 200.0) / 200.0, 0.0, 1.0)
		var slowdown: float = pow(0.5, delta * far_amount)
		comet_velocity *= slowdown

		var to_hole: Vector2 = hole_position - comet_position
		var hole_distance: float = maxf(to_hole.length(), 60.0)
		var hole_pull: float = minf(1600.0, 2000000.0 / (hole_distance * hole_distance))
		comet_velocity += to_hole.normalized() * hole_pull * delta
		var from_center: Vector2 = comet_position - hole_position
		var edge_fraction: float = maxf(absf(from_center.x) / 960.0, absf(from_center.y) / 540.0)
		var edge_depth: float = maxf((edge_fraction - 0.8) / 0.2, 0.0)
		var outward: Vector2 = from_center.normalized()
		var outward_speed: float = maxf(comet_velocity.dot(outward), 0.0)

		var border_pull: float = edge_depth * (150.0 + outward_speed * 3.0)
		border_pull += 700.0 * edge_depth * edge_depth
		comet_velocity -= outward * minf(border_pull, 6000.0) * delta

		comet_position += comet_velocity * delta
		bounce_off_planet()
		orbit_around_hole()
		assist_planet_orbit(delta)

func _unhandled_input(event: InputEvent) -> void:
	if event is InputEventKey and event.pressed and not event.echo:
		if event.keycode == KEY_ESCAPE:
			show_screen("menu")
			return
		if event.keycode == KEY_R and screen in ["play", "lost", "won"]:
			start_run()
			return
	if screen != "play":
		return
	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_LEFT:
			if event.pressed:
				drag_start = event.position
				is_aiming = true
			elif is_aiming:
				is_aiming = false

				var shot_velocity: Vector2 = ((drag_start - event.position) * 4.0).limit_length(630.0)

				if shot_velocity.length() < 60.0:
					return

				var power: float = shot_velocity.length() / 630.0
				var cost: float = 10.0 + 20.0 * power

				if energy < cost:
					$EnergyLabel.text = "Not enough energy for this shot"
					return

				energy -= cost
				comet_velocity = shot_velocity
				has_launched = true
				hot_time = 3.0

func bounce_off_planet() -> void:
	for planet in planets:
		var planet_node: Sprite2D = planet["node"]
		var radius: float = planet["radius"]
		var away: Vector2 = comet_position - planet_node.position
		var contact_distance: float = radius + 20.0

		if away.length() < contact_distance:
			var normal: Vector2 = away.normalized()

			if away.length_squared() == 0.0:
				normal = Vector2.RIGHT

			comet_position = planet_node.position + normal * contact_distance

			var inward_speed: float = comet_velocity.dot(normal)
			if inward_speed < 0.0:
				comet_velocity -= 1.9 * inward_speed * normal

func orbit_around_hole() -> void:
	var away: Vector2 = comet_position - hole_position
	var safe_radius: float = 110.0

	if away.length() >= safe_radius:
		return

	var outward: Vector2 = away.normalized()
	if away.length_squared() == 0.0:
		outward = Vector2.RIGHT

	var turn: float = 1.0
	if away.cross(comet_velocity) < 0.0:
		turn = -1.0

	comet_position = hole_position + outward * safe_radius

	var sideways: Vector2 = Vector2(-outward.y, outward.x) * turn
	var orbit_speed: float = clampf(comet_velocity.length(), 120.0, 220.0)
	comet_velocity = sideways * orbit_speed

func warp_point(point: Vector2) -> Vector2:
	var toward_hole: Vector2 = hole_position - point
	var distance: float = toward_hole.length()
	var bend: float = 168.0 * exp(-distance / 408.0)
	bend = minf(bend, distance * 0.8)
	return point + toward_hole.normalized() * bend

func assist_planet_orbit(delta: float) -> void:
	if comet_velocity.length() > 220.0:
		return

	for planet in planets:
		var planet_node: Sprite2D = planet["node"]
		var radius: float = planet["radius"]
		var away: Vector2 = comet_position - planet_node.position
		var distance: float = away.length()

		if distance < 1.0 or distance > radius * 3.0:
			continue

		var outward: Vector2 = away.normalized()
		var turn: float = 1.0
		if away.cross(comet_velocity) < 0.0:
			turn = -1.0

		var sideways: Vector2 = Vector2(-outward.y, outward.x) * turn

		var size_factor: float = radius / 60.0
		var strength: float = gravity_strength * size_factor * size_factor
		var orbit_speed: float = clampf(sqrt(strength / distance), 70.0, 150.0)
		var target_velocity: Vector2 = sideways * orbit_speed

		if distance < radius + 40.0:
			target_velocity += outward * 60.0

		comet_velocity = comet_velocity.lerp(target_velocity, minf(2.0 * delta, 1.0))
		return
